#!/usr/bin/env python3
"""A1 모집단 입력 · A3 바탕 · A4 설치 전 파일을 정확한 바이트로 쓴다."""
import os, sys

out = sys.argv[1]
os.makedirs(out, exist_ok=True)

def w(name, text):
    with open(os.path.join(out, name), "wb") as f:
        f.write(text.encode("utf-8"))

indent2 = '{\n  "model": "opus",\n  "permissions": {\n    "allow": [\n      "Bash(ls)"\n    ]\n  }\n}\n'
w("a1-01-indent2.json", indent2)
w("a1-02-indent4.json", '{\n    "env": {\n        "A": "1"\n    },\n    "model": "opus"\n}\n')
w("a1-03-tab.json", '{\n\t"model": "opus",\n\t"permissions": {\n\t\t"deny": [\n\t\t\t"Bash(rm)"\n\t\t]\n\t}\n}\n')
w("a1-04-one-line.json", '{"model": "opus", "permissions": {"allow": ["Bash(ls)"]}}\n')
w("a1-05-crlf.json", indent2.replace("\n", "\r\n"))
w("a1-06-no-final-newline.json", indent2[:-1])
w("a1-07-empty-object.json", "{}\n")
w("a1-08-same-event-other-hook.json",
  '{\n  "hooks": {\n    "Stop": [\n      {\n        "hooks": [\n          {\n            "type": "command",\n            "command": "notify.sh"\n          }\n        ]\n      }\n    ]\n  }\n}\n')

  '{\n  "hooks": {\n    "PreToolUse": [\n      {\n        "matcher": "Bash",\n        "hooks": [\n          {\n            "type": "command",\n            "command": "guard.sh"\n          }\n        ]\n      }\n    ]\n  },\n  "model": "opus"\n}\n')
w("a1-10-no-hooks.json", '{\n  "env": {\n    "DEBUG": "1"\n  },\n  "includeCoAuthoredBy": false\n}\n')
w("a1-11-empty-event-array.json", '{"hooks":{"Stop":[]}}\n')
w("a1-12-two-empty-event-arrays-and-other-key.json",
  '{\n  "hooks": {\n    "Stop": [],\n    "SubagentStop": []\n  },\n  "model": "opus"\n}\n')
w("a1-13-key-order.json",
  '{\n  "permissions": {\n    "allow": [\n      "Read"\n    ]\n  },\n  "model": "opus",\n  "env": {\n    "Z": "1",\n    "A": "2"\n  }\n}\n')
w("a1-14-escapes-and-numbers.json",
  '{\n  "hooks": {\n    "PostToolUse": [\n      {\n        "hooks": [\n          {\n            "type": "command",\n            "command": "\\u0041 \\/tmp\\/x.sh",\n            "timeout": 1.0\n          }\n        ],\n        "weight": 1e3\n      }\n    ]\n  }\n}\n')

  '{\n  "hooks": {\n    "Stop": [{"hooks": [{"type": "command", "command": "notify.sh"}]}]\n  },\n  "model": "opus"\n}\n')
w("a1-16-empty-hooks-object.json", '{\n  "hooks": {},\n  "model": "opus"\n}\n')
w("a1-17-bare-empty-object.json", "{}")
w("a1-18-crlf-indent4-no-final-newline.json",
  '{\r\n    "hooks": {\r\n        "Stop": [\r\n            {\r\n                "hooks": [\r\n                    {\r\n                        "type": "command",\r\n                        "command": "notify.sh"\r\n                    }\r\n                ]\r\n            }\r\n        ]\r\n    },\r\n    "model": "opus"\r\n}')
w("a1-19-compact-no-spaces.json", '{"model":"opus","hooks":{"SubagentStop":[{"hooks":[{"type":"command","command":"x.sh"}]}]}}')

# A3 — 사용자가 설치 뒤에 고치는 방의 바탕.
w("a3-base.json",
  '{\n  "model": "opus",\n  "env": {\n    "A": "1"\n  },\n  "hooks": {\n    "Stop": [\n      {\n        "hooks": [\n          {\n            "type": "command",\n            "command": "notify.sh"\n          }\n        ]\n      }\n    ]\n  },\n  "permissions": {\n    "allow": [\n      "Bash(ls)"\n    ]\n  }\n}\n')

# A4 — 옛 설치(재직렬화) 전의 사용자 파일. 재직렬화하면 바이트가 달라지는 형태.

  '{\n    "permissions": {\n        "allow": ["Bash(ls)"]\n    },\n    "model": "opus"\n}\n')

#!/bin/bash
# A2 골든 · A4 옛 설치 fixture 를 착수 커밋(acd7e82) 빌드로 뜬다.
set -euo pipefail
S=/private/tmp/claude-501/-Users-incognito-dev-projects-palimpsest/c7c1de0a-52f2-48d8-b5d1-8926cb99dbb6/scratchpad
OLD=$S/acd7e82-target/debug/pal
FX=/Users/incognito/dev/projects/palimpsest/.claude/worktrees/agent-aa351aeafb4b52cbd/crates/pal-cli/tests/fixtures/clean_uninstall/settings
echo "착수 빌드: $($OLD --version)  ($OLD)"
echo "착수 worktree HEAD: $(git -C $S/acd7e82 rev-parse HEAD)"
ROOMS=$S/golden-rooms
rm -rf "$ROOMS"
mkdir -p "$ROOMS"

방() {
  local r=$1
  mkdir -p "$r/.claude" "$r/home"
  git -C "$r" init -q .
  git -C "$r" config core.autocrlf false
  echo hi > "$r/README.md"
}

for f in "$FX"/a1-*.json; do
  case "$f" in *.golden.json) continue;; esac
  name=$(basename "$f" .json)
  r=$ROOMS/$name
  방 "$r"
  cp "$f" "$r/.claude/settings.json"
  (cd "$r" && HOME=$r/home XDG_DATA_HOME=$r/home/.local/share "$OLD" install >/dev/null)
  cp "$r/.claude/settings.json" "$FX/$name.golden.json"
  echo "골든  $name.golden.json  sha256=$(shasum -a 256 "$FX/$name.golden.json" | cut -d' ' -f1)"
done

# A4 — 설치 전 파일을 커밋한 방에 착수 빌드로 설치하고, 설정 쪽 기록을 뜬다.
r=$ROOMS/a4
방 "$r"
cp "$FX/a4-old-install/settings.before.json" "$r/.claude/settings.json"
git -C "$r" add -A
git -C "$r" -c user.email=t@example.com -c user.name=t commit -qm before
(cd "$r" && HOME=$r/home XDG_DATA_HOME=$r/home/.local/share "$OLD" install >/dev/null)
D=$FX/a4-old-install
cp "$r/.claude/settings.json" "$D/settings.installed.json"
cp "$r/.claude/pal/manifest.json" "$D/manifest.json"
cp "$r/CLAUDE.md" "$D/CLAUDE.md.txt"
cp "$r/.gitignore" "$D/gitignore.txt"
cp "$r/.palimpsest/manifest.toml" "$D/manifest.toml.txt"
for x in settings.installed.json manifest.json CLAUDE.md.txt gitignore.txt manifest.toml.txt; do
  echo "A4    a4-old-install/$x  sha256=$(shasum -a 256 "$D/$x" | cut -d' ' -f1)"
done
